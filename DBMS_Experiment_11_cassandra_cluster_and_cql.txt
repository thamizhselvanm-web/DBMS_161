/* ============================================================================
   EX NO: 11
   TITLE : DISTRIBUTED DATABASE CLUSTER USING APACHE CASSANDRA & CQL
   AIM   : To configure a multi-node Apache Cassandra distributed cluster,
           verify ring topology via nodetool, and perform keyspace and table
           operations using Cassandra Query Language (CQL).
   ENGINE: Apache Cassandra 4.x/5.x / cqlsh
   ============================================================================
   ALGORITHM:
   ---------
   STEP 1: Start.
   STEP 2: Configure cluster networking in `cassandra.yaml` on all nodes:
           - `cluster_name`: 'CollegeCluster'
           - `seeds`: "192.168.1.10"
           - `listen_address`: Host machine IP
           - `endpoint_snitch`: GossipingPropertyFileSnitch
   STEP 3: Start the Cassandra daemon service across cluster nodes.
   STEP 4: Verify distributed cluster topology and node health using `nodetool status`.
   STEP 5: Launch interactive CQL shell (`cqlsh`).
   STEP 6: Create keyspace `college_ks` with `SimpleStrategy` and replication factor 3.
   STEP 7: Switch to keyspace and create columnar table `student`.
   STEP 8: Execute CQL DML insertions, row selections, and targeted updates.
   STEP 9: Verify query outputs and conclude the experiment.
   ============================================================================ */


# ============================================================================
# PART 1: CLUSTER CONFIGURATION (cassandra.yaml on every node)
# ============================================================================

cluster_name: 'CollegeCluster'
seeds: "192.168.1.10"
listen_address: 192.168.1.10
rpc_address: 192.168.1.10
endpoint_snitch: GossipingPropertyFileSnitch


# ============================================================================
# PART 2: SERVICE INITIALIZATION & CLUSTER STATUS VERIFICATION
# ============================================================================

# Start the Cassandra service daemon
sudo service cassandra start

# Verify multi-node cluster status and gossip ring
nodetool status

# EXPECTED OUTPUT:
# Datacenter: datacenter1
# =======================
# Status=Up/Down
# |/ State=Normal/Leaving/Joining/Moving
# --  Address       Load       Tokens  Owns (effective)  Host ID                               Rack
# UN  192.168.1.10  95.42 KB   256     34.0%             a1b2c3d4-e5f6-7890-abcd-ef1234567890  rack1
# UN  192.168.1.11  102.15 KB  256     33.0%             b2c3d4e5-f6a7-8901-bcde-fa2345678901  rack1
# UN  192.168.1.12  98.67 KB   256     33.0%             c3d4e5f6-a7b8-9012-cdef-ab3456789012  rack1
#
# (Note: 'UN' indicates status is Up and state is Normal)


# ============================================================================
# PART 3: CQL SHELL & KEYSPACE OPERATIONS
# ============================================================================

# Open CQL interactive shell
cqlsh 192.168.1.10 9042

# Create Keyspace with SimpleStrategy replication
CREATE KEYSPACE college_ks
WITH REPLICATION = {
    'class': 'SimpleStrategy',
    'replication_factor': 3
};

# Switch context to the created keyspace
USE college_ks;


# ============================================================================
# PART 4: TABLE CREATION (DDL)
# ============================================================================

CREATE TABLE student (
    rollno INT PRIMARY KEY,
    name TEXT,
    dept TEXT
);


# ============================================================================
# PART 5: DATA INSERTION (DML)
# ============================================================================

INSERT INTO student (rollno, name, dept) VALUES (101, 'RAJA', 'CSE');
INSERT INTO student (rollno, name, dept) VALUES (102, 'PRIYA', 'IT');
INSERT INTO student (rollno, name, dept) VALUES (103, 'KARTHIK', 'ECE');


# ============================================================================
# PART 6: DATA RETRIEVAL (CQL QUERY)
# ============================================================================

SELECT * FROM student;

# EXPECTED OUTPUT:
#  rollno | dept | name
# --------+------+---------
#     101 |  CSE | RAJA
#     102 |   IT | PRIYA
#     103 |  ECE | KARTHIK
#
# (3 rows)


# ============================================================================
# PART 7: PARTITION KEY FILTERING & UPDATE
# ============================================================================

SELECT * FROM student WHERE rollno = 101;

# EXPECTED OUTPUT:
#  rollno | dept | name
# --------+------+------
#     101 |  CSE | RAJA
#
# (1 rows)

UPDATE student SET dept = 'AI-DS' WHERE rollno = 101;

SELECT * FROM student WHERE rollno = 101;

# EXPECTED OUTPUT:
#  rollno | dept  | name
# --------+-------+------
#     101 | AI-DS | RAJA
#
# (1 rows)


# ============================================================================

# The distributed Apache Cassandra cluster was initialized, inspected via
# nodetool status, and CQL keyspace, table, and row queries were verified.
